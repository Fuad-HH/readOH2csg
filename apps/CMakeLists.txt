add_library(omegah2csg_cli STATIC cli.cpp)
target_link_libraries(omegah2csg_cli PUBLIC omegah2csg_core)

if(DEFINED SKBUILD)
  set(omegah2csg_bin_dir ${SKBUILD_PROJECT_NAME}/bin)
else()
  set(omegah2csg_bin_dir ${CMAKE_INSTALL_BINDIR})
endif()

foreach(app openmc degas2)
  string(TOUPPER ${app} APP)
  set(exe ${OMEGAH2CSG_${APP}_EXE_NAME})
  add_executable(${exe} ${app}_main.cpp)
  target_link_libraries(${exe} PRIVATE omegah2csg_cli)
  set_target_properties(${exe} PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
  install(TARGETS ${exe} RUNTIME DESTINATION ${omegah2csg_bin_dir})
endforeach()
